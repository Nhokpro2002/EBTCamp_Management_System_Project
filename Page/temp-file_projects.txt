<script>

    /* =====================================================
       CONSTANTS
       ===================================================== */

    const COLLECTION_PROJECTS = "Projects";
    const COLLECTION_PROBLEMS = "Problems";
    const COLLECTION_TASKS = "Tasks";
    const POCKETBASE_URL = "http://127.0.0.1:8090";
    let projectData = {};

    let projectProcess = 0; // (taskDone/totalTask) * 100


    /* =====================================================
       PAGE INITIALIZATION
       ===================================================== */

    $(document).ready(function () {

        // process focus option sidebar

        initSidebar();

        initChart();

        getAllProjectData();


        // Based on the number of projects, render the project-item tags accordingly.
    })

    /* =====================================================
       SIDEBAR
       ===================================================== */

    function initSidebar() {
        const currentPage = window.location.pathname;

        if (currentPage.includes('dashboard')) {
            document.getElementById('dashboard-link').classList.add('active');
        } else if (currentPage.includes('orders')) {
            document.getElementById('orders-link').classList.add('active');
        } else if (currentPage.includes('orders')) {
            document.getElementById('users-link').classList.add('active');
        } else if (currentPage.includes('orders')) {
            document.getElementById('settings-link').classList.add('active');
        } else if (currentPage.includes('projects')) {
            document.getElementById('projects-link').classList.add('active');
        }
    }

    /* =====================================================
       CHART
       ===================================================== */

    function initChart() {
        const ctx = document.getElementById('projectChart');

        new Chart(ctx, {
            type: 'bar',
            data: {
                labels: [
                    'Week 1',
                    'Week 2',
                    'Week 3',
                    'Week 4',
                    'Week 5'
                ],
                datasets: [
                    {
                        type: 'line',
                        label: 'Planned',
                        data: [20, 40, 60, 80, 100],
                        borderColor: '#0d6efd',
                        tension: .4
                    },
                    {
                        type: 'line',
                        label: 'Actual',
                        data: [15, 35, 55, 70, 85],
                        borderColor: '#28a745',
                        tension: .4
                    },
                    {
                        label: 'Completed Tasks',
                        data: [5, 10, 15, 20, 25],
                        backgroundColor: '#cfe2ff'
                    }
                ]
            },
            options: {
                responsive: true
            }
        });
    }


    /* =====================================================
       PROJECT APIs
       ===================================================== */

    /*
 Fetch all projects from PocketBase.

 After data is loaded:
 1. Render project list
 2. Auto focus first project
 3. Fill KPI cards
*/
    async function getAllProjectData() {
        try {
            const token = localStorage.getItem("token");
            const response = await fetch(`${POCKETBASE_URL}/api/collections/${COLLECTION_PROJECTS}/records`, {
                headers: {
                    'Authorization': `Bearer ${token}`,
                    'Content-Type': 'application/json'
                }
            });

            if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
            const result = await response.json();
            projectData = result;

            // call renderProjectItem
            renderProjectItem(projectData);

            // call fillAreaOverallProcess
            fillAreaOverallProcess(projectData.items[0].id);

            fillAreaTotalTasks(projectData.items[0].id);



        } catch (error) {
            console.log(error);
        }
    }

    /* =====================================================
       PROJECT UI RENDER
       ===================================================== */
    /*
 Render project cards on left panel.

 Each card:
 - Displays project name
 - Allows user to switch project
 - Refreshes KPI data
*/
    function renderProjectItem(projectData) {
        const projectList = document.getElementById("projectList");
        projectList.innerHTML = "";
        for (let index = 0; index < projectData.items.length; index++) {

            const div = document.createElement("div");
            div.className = "project-item active";
            const item = projectData.items[index];

            div.innerHTML = `
            <div class="fw-bold">${projectData.items[index].project_name}</div>
            <div class="progress mt-2">        
                <div class="progress-bar bg-success" style="width:${10}%">
                                        </div>   
                </div>
            </div>
        `;

            projectList.appendChild(div);

            // Bind onclick event
            div.addEventListener("click", function () {
                div.classList.add("active");
                fillAreaOverallProcess(item.id);
                fillAreaTotalTasks(item.id);

            });
        }

    }

    /* =====================================================
       KPI AREA
       ===================================================== */

    /*
    projectID: ID of project that is being focused on
    Fill percent process project (%)
    Count tasks are done
    Percent = tasks done/total tasks
    */
    async function fillAreaOverallProcess(projectID) {

        try {
            const token = localStorage.getItem("token");
            const response = await fetch(`${POCKETBASE_URL}/api/collections/${COLLECTION_TASKS}/records?filter=project="${projectID}"`, {
                headers: {
                    'Authorization': `Bearer ${token}`,
                    'Content-Type': 'application/json'
                }
            });

            if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
            const result = await response.json();
            let countTasksDone = 0;
            let totalTasks = result.items.length;

            for (let index = 0; index < result.items.length; index++) {
                if (result.items[index].status == "Done") {
                    countTasksDone++;
                }
            }

            const percent = Math.floor((countTasksDone / totalTasks) * 100);
            projectProcess = percent;
            const h1Element = document.querySelector("#area-overall-process h1");
            h1Element.textContent = `${percent}%`;

            const progressBar = document.querySelector("#area-overall-process .progress-bar");

            // Change the width to 50%
            progressBar.style.width = `${percent}%`;


        } catch (error) {
            console.log(error);
        }

    }

    /*
    Count task status for selected project.

    Display:
    - Total tasks
    - Done tasks
    - Processing tasks
    - Pending tasks
*/
    async function fillAreaTotalTasks(projectID) {

        try {
            const token = localStorage.getItem("token");
            const response = await fetch(`${POCKETBASE_URL}/api/collections/${COLLECTION_TASKS}/records?filter=project="${projectID}"`, {
                headers: {
                    'Authorization': `Bearer ${token}`,
                    'Content-Type': 'application/json'
                }
            });

            if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
            const result = await response.json();
            let countTasksDone = 0;
            let countTasksProcessing = 0;
            let countTasksPending = 0;
            let totalTasks = result.items.length;

            for (let index = 0; index < result.items.length; index++) {
                if (result.items[index].status == "Done") {
                    countTasksDone++;
                } else if (result.items[index].status == "Pending") {
                    countTasksPending++;
                } else if (result.items[index].status == "Processing") {
                    countTasksProcessing++;
                }
            }

            const h1TotalTasks = document.querySelector("#area-total-tasks h1");
            h1TotalTasks.textContent = `${totalTasks}`;
            document.querySelector('.done .status-value').textContent = `${countTasksDone}`;
            document.querySelector('.processing .status-value').textContent = `${countTasksProcessing}`;
            document.querySelector('.pending .status-value').textContent = `${countTasksPending}`;

        } catch (error) {
            console.log(error);
        }

        try {
            const token = localStorage.getItem("token");
            const response = await fetch(`${POCKETBASE_URL}/api/collections/${COLLECTION_PROBLEMS}/records?filter=project="${projectID}"`, {
                headers: {
                    'Authorization': `Bearer ${token}`,
                    'Content-Type': 'application/json'
                }
            });

            if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
            const result = await response.json();
            let totalIssues = result.items.length;

            document.querySelector('.issue .status-value').textContent = `${totalIssues}`;

        } catch (error) {
            console.log(error);
        }


    }
